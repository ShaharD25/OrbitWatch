cmake_minimum_required(VERSION 3.20)
project(OrbitWatch VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ORBITWATCH_BUILD_GUI "Build Qt desktop application and JSON CLI" ON)
option(ORBITWATCH_BUILD_3D "Build GPLv3 Qt Quick 3D Earth view" ON)
option(ORBITWATCH_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(orbitwatch_core src/core.cpp)
target_include_directories(orbitwatch_core PUBLIC include)
target_compile_options(orbitwatch_core PRIVATE -Wall -Wextra -Wpedantic)
if(ORBITWATCH_SANITIZERS)
  target_compile_options(orbitwatch_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(orbitwatch_core PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(orbitwatch_tests tests/core_tests.cpp)
  target_link_libraries(orbitwatch_tests PRIVATE orbitwatch_core)
  add_test(NAME core COMMAND orbitwatch_tests)
  add_executable(orbitwatch_benchmark tests/benchmark.cpp)
  target_link_libraries(orbitwatch_benchmark PRIVATE orbitwatch_core)
endif()
if(ORBITWATCH_BUILD_GUI)
  find_package(Qt6 REQUIRED COMPONENTS Core Widgets Concurrent)
  add_library(orbitwatch_io src/io.cpp)
  target_link_libraries(orbitwatch_io PUBLIC orbitwatch_core Qt6::Core)
  add_library(orbitwatch_gui src/window.cpp)
  target_link_libraries(orbitwatch_gui PUBLIC orbitwatch_io Qt6::Widgets Qt6::Concurrent)
  if(ORBITWATCH_BUILD_3D)
    find_package(Qt6 6.5 REQUIRED COMPONENTS QuickWidgets Quick3D)
    target_sources(orbitwatch_gui PRIVATE src/earth_view.cpp)
    target_compile_definitions(orbitwatch_gui PUBLIC ORBITWATCH_HAS_3D)
    target_link_libraries(orbitwatch_gui PUBLIC Qt6::QuickWidgets Qt6::Quick3D)
    qt_add_resources(orbitwatch_gui earth PREFIX "/earth" FILES
      qml/Earth.qml qml/earth-surface.frag qml/atmosphere.vert qml/atmosphere.frag assets/earth-day.jpg assets/earth-clouds.jpg assets/earth-night.jpg)
  endif()
  add_executable(orbitwatch main.cpp)
  target_link_libraries(orbitwatch PRIVATE orbitwatch_gui)
  add_executable(orbitwatch_cli src/cli.cpp)
  target_link_libraries(orbitwatch_cli PRIVATE orbitwatch_io)
  if(BUILD_TESTING)
    add_executable(orbitwatch_io_tests tests/io_tests.cpp)
    target_link_libraries(orbitwatch_io_tests PRIVATE orbitwatch_io)
    add_test(NAME io COMMAND orbitwatch_io_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      add_test(NAME cli_e2e COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py $<TARGET_FILE:orbitwatch_cli> ${CMAKE_CURRENT_SOURCE_DIR})
      set_tests_properties(cli_e2e PROPERTIES TIMEOUT 60)
    endif()
    add_executable(orbitwatch_gui_tests tests/gui_tests.cpp)
    target_link_libraries(orbitwatch_gui_tests PRIVATE orbitwatch_gui)
    add_test(NAME gui_flow COMMAND orbitwatch_gui_tests)
    set_tests_properties(gui_flow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
    if(ORBITWATCH_BUILD_3D)
      find_package(Qt6 REQUIRED COMPONENTS Test)
      add_executable(orbitwatch_earth_tests tests/earth_tests.cpp)
      target_link_libraries(orbitwatch_earth_tests PRIVATE orbitwatch_gui Qt6::Test)
      add_test(NAME earth_controls COMMAND orbitwatch_earth_tests)
      set_tests_properties(earth_controls PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
    endif()
    add_test(NAME gui_smoke COMMAND orbitwatch --smoke-test)
    set_tests_properties(gui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
  endif()
endif()
